% DEFINICIÓN COMPLETA DEL ESTILO DE CITA APA PARA gbpublisher.
% ARCHIVO AUTOCONTENIDO: CARGA EL PAQUETE Y HACE SUS AJUSTES.
%
% LO CONSUMEN LAS DOS RAMAS:
%   revistas — \input con ruta absoluta a ~/.gbpublisher/latex/
%   libros   — copiado a {proyecto}/latex/, \input por nombre
%
% POR ESO EL \usepackage VA ACÁ Y NO EN EL PREÁMBULO QUE LO
% LLAMA: LAS OPCIONES style, backend Y sorting SON PARTE DE LA
% DEFINICIÓN DEL ESTILO, NO DEL DOCUMENTO. SI QUEDARAN AFUERA
% HABRÍA DOS LUGARES DONDE MIRAR CUANDO UNA CITA SALE DISTINTA.
%
% PRECONDICIÓN: babel Y csquotes YA CARGADOS. ES EL ORDEN QUE
% PIDE LA DOCUMENTACIÓN DE biblatex.
%
% NOTA: style=apa VIENE DEL PAQUETE biblatex-apa, QUE ESTÁ EN
% TeX Live. ESTE ARCHIVO SON NUESTROS AJUSTES SOBRE UN ESTILO
% DE TeX Live, NO UN ESTILO PROPIO.
% ============================================================

\usepackage[
  style=apa,
  sortcites=true,
  citereset=chapter,
  backend=biber,
  backref=false,
  indexing=true
]{biblatex}
\DeclareLanguageMapping{spanish}{spanish-apa}

\renewbibmacro*{bibindex}{%
	\ifbibindex
	{\indexnames{author}\indexnames{editor}\indexnames{translator}}
	{}}
\renewbibmacro*{citeindex}{%
	\ifciteindex
	{\indexnames{labelname}}
	{}}

% --- Sangría de referencias ---
\setlength{\bibhang}{2em}

% --- Definiciones para el tratamiento de las url ---
\defcounter{biburlnumpenalty}{3000}
\defcounter{biburlucpenalty}{6000}
\defcounter{biburllcpenalty}{9000}

% --- STRINGS PROPIOS PARA TIPOS DE EDITOR ---
% biblatex-apa NO TRAE comp., coord., dir., col. NI org., QUE
% SON DE USO CORRIENTE EN LA EDICIÓN ACADÉMICA EN CASTELLANO.
\NewBibliographyString{typecompiler}
\NewBibliographyString{typecompilers}
\NewBibliographyString{typecoordinator}
\NewBibliographyString{typecoordinators}
\NewBibliographyString{typedirector}
\NewBibliographyString{typedirectors}
\NewBibliographyString{typecollaborator}
\NewBibliographyString{typecollaborators}
\NewBibliographyString{typeorganizer}
\NewBibliographyString{typeorganizers}

\DefineBibliographyStrings{spanish}{%
  typecompiler      = {comp\adddot},
  typecompilers     = {comps\adddot},
  typecoordinator   = {coord\adddot},
  typecoordinators  = {coords\adddot},
  typedirector      = {dir\adddot},
  typedirectors     = {dirs\adddot},
  typecollaborator  = {col\adddot},
  typecollaborators = {cols\adddot},
  typeorganizer     = {org\adddot},
  typeorganizers    = {orgs\adddot},
}
