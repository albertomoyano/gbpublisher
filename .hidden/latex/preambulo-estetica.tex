% OBJETIVO GENERAL:
% - Personalizar la apariencia de partes, capítulos, secciones,

% ============================================================
% DISEÑO DE PARTES
% ------------------------------------------------------------
% Redefinición de \@part y \@spart para controlar:
% - Numeración de partes (si corresponde)
% - Entrada en el TOC
% - Centrado del título y subtítulo de la parte
% - Tamaño de fuente y estilo tipográfico
%
% \@part[#1]#2:
%   - #1: título corto para TOC y encabezado
%   - #2: título completo para mostrar en la página
%   - Centrado, fuente sans serif (\sf), \LARGE para nombre de parte
%     y \Large para título completo
%
% \@spart#1:
%   - Para partes no numeradas
%   - Solo muestra el título centrado, \LARGE, sans serif
%
\makeatletter
\def\@part[#1]#2{%
	\ifnum \c@secnumdepth >-2\relax
	\refstepcounter{part}%
	\addcontentsline{toc}{part}{Parte \thepart\hspace{1em}#1}%
	\else
	\addcontentsline{toc}{part}{#1}%
	\fi
	\markboth{}{}%
	{\centering
		\interlinepenalty \@M
		\ifnum \c@secnumdepth >-2\relax
		\sf\LARGE\selectfont \partname~\thepart
		\par\vskip 20\p@%
		\fi
		\sf\Large\selectfont
		#2\par}%
	\@endpart
}

\def\@spart#1{%
	\markboth{}{}%
	{\centering
		\interlinepenalty \@M
		\sf\LARGE\selectfont
		#1\par}%
	\@endpart
}
\makeatother

% ASÍ QUE NO SIRVE PARA UNA CADENA CON pdfTeX.
% ============================================================
\usepackage{linebreaker}
\linebreakersetup{
	maxtolerance=90,
	maxemergencystretch=1em,
	maxcycles=4
}

% ============================================================
% NUMERACIÓN ROMANA DE PRELIMINARES EN VERSALITAS
% ------------------------------------------------------------
% La clase book las pone en minúscula, que es la convención
% anglosajona. En español el foliado romano va en mayúscula.
\renewcommand{\thepage}{%
  \ifnum\value{page}>0
    \ifnum\thechapter>0 \arabic{page}\else \Roman{page}\fi
  \fi}

\makeatletter
% \frontmatter DE book HACE \pagenumbering{roman}. SE REDEFINE EL FORMATO DEL
% CONTADOR PARA QUE USE VERSALITAS, QUE ES LA CONVENCIÓN EN ESPAÑOL.
%
% NO SE USA \uppercase: NO SE EXPANDE AL ESCRIBIR EL .idx NI EL .aux, ASÍ QUE
% AHÍ QUEDA EL COMANDO LITERAL. xindy NO LO RECONOCE COMO PÁGINA Y DESCARTA LA
% ENTRADA CON UN AVISO QUE EL COMPILADOR TRAGA.
%
% TAMPOCO \Roman{#1}: \@roman RECIBE UN NÚMERO, NO EL NOMBRE DE UN CONTADOR.
% \@Roman ES LA MACRO INTERNA QUE LaTeX YA TIENE PARA ESTO.
\renewcommand{\@roman}[1]{\@Roman{#1}}
\makeatother

% ============================================================
% CABEZALES Y PIES DE PÁGINA
% ------------------------------------------------------------
% titleps viene con titlesec: requiere la opción «pagestyles»,
% que ya está en la línea del \usepackage.
%
% El título y la autoría los pone el generador antes de cada
% \include con \gbMarcasPieza. No se usan \chaptertitle —que
% trae el título completo— ni \Author, que había que declarar a
% mano en cada pieza.
% ============================================================

% ESTILO plain: ES EL QUE LaTeX APLICA EN LA PRIMERA PÁGINA DE
% CADA CAPÍTULO. SE VACÍA PARA QUE ESAS PÁGINAS NO LLEVEN FOLIO.
\renewpagestyle{plain}[]{%
  \setfoot{}{}{}}

% ESTILO GENERAL
%   par   folio — autoría
%   impar título — folio
\newpagestyle{myps}[]{%
  \setfoot[][][]{}{}{}%
  \sethead
  [\sffamily\textbf{\usepage}][][\sffamily\gbAutoriaFolio]
  {\sffamily\gbTituloFolio}{}{\sffamily\textbf{\usepage}}}

\pagestyle{myps}
